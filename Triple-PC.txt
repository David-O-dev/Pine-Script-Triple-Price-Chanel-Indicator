//@version=6
indicator("Triple Price Channel", overlay = true, precision = 5)

//PC 1
PClength1 = input(9, group = "PC 1")
offsetlength1 = input(0, group = "PC 1")
PCColor1 = input.color(defval = color.black, title = "Price Channel 1 Color", group = "PC 1")

highprice1 = ta.highest(high, PClength1)
lowprice1 = ta.lowest(low, PClength1)
centerline1 = (ta.highest(high, PClength1) +  ta.lowest(low, PClength1)) / 2
plot(highprice1, color=PCColor1, title="Highprice Line 1", offset=offsetlength1, linewidth = 4)
plot(lowprice1, color=PCColor1, title="Lowprice Line 1", offset=offsetlength1, linewidth = 4)
